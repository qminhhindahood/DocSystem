```typescript
/**
 * Call LM Studio OpenAI-compatible chat completions endpoint (non-streaming)
 */
async function callLMStudio(
  messages: Array<{ role: string; content: string }>,
  options?: { temperature?: number; max_tokens?: number; num_ctx?: number }
): Promise<string> {
  const response = await withRetry(
    () => axios.post(
      `${LM_STUDIO_URL}/v1/chat/completions`,
      {
        model: LM_STUDIO_MODEL,
        messages,
        temperature: options?.temperature ?? 0.3,
        max_tokens: options?.max_tokens ?? 4096,
      },
      { timeout: 120000 },
    ),
    { maxRetries: 2, baseDelay: 1000, retryContext: 'lmstudio-chat' },
  );
  return response.data.choices?.[0]?.message?.content || '';
}

/**
 * Call LM Studio with streaming (returns async generator of text chunks)
 */
async function* streamLMStudio(
  messages: Array<{ role: string; content: string }>,
  options?: { temperature?: number; max_tokens?: number }
): AsyncGenerator<string> {
  const response = await withRetry(
    () => axios.post(
      `${LM_STUDIO_URL}/v1/chat/completions`,
      {
        model: LM_STUDIO_MODEL,
        messages,
        stream: true,
        temperature: options?.temperature ?? 0.3,
        max_tokens: options?.max_tokens ?? 4096,
      },
      { responseType: 'stream', timeout: 180000 },
    ),
    { maxRetries: 1, baseDelay: 2000, retryContext: 'lmstudio-stream' },
  );

  let buffer = '';
  for await (const chunk of response.data) {
    buffer += chunk.toString();
    const lines = buffer.split('\n');
    buffer = lines.pop() || '';

    for (const line of lines) {
      const trimmed = line.trim();
      if (!trimmed || trimmed === 'data: [DONE]') continue;
      if (!trimmed.startsWith('data: ')) continue;

      try {
        const data = JSON.parse(trimmed.slice(6));
        const content = data.choices?.[0]?.delta?.content;
        if (content) yield content;
      } catch {
        // skip malformed JSON lines
      }
    }
  }
  // Process remaining buffer
  if (buffer.trim() && buffer.trim() !== 'data: [DONE]') {
    try {
      const data = JSON.parse(buffer.trim().startsWith('data: ')
        ? buffer.trim().slice(6)
        : buffer.trim());
      const content = data.choices?.[0]?.delta?.content;
      if (content) yield content;
    } catch {
      // ignore
    }
  }
}
```
